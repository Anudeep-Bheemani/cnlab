#include <stdio.h>

int main()
{
    int n, w, base = 0, i, lost;

    printf("Enter number of frames: ");
    scanf("%d", &n);

    printf("Enter window size: ");
    scanf("%d", &w);

    if (n <= 0 || w <= 0 || w > n)
    {
        printf("Invalid input\n");
        return 1;
    }

    while (base < n)
    {
        int end = base + w;

        if (end > n)
            end = n;

        printf("\nSending frames:\n");

        for (i = base; i < end; i++)
            printf("Frame %d sent\n", i);

        printf("Enter lost frame number in this window (-1 for none): ");
        scanf("%d", &lost);

        if (lost >= base && lost < end)
        {
            printf("\nFrame %d lost!\n", lost);
            printf("Receiver discards frames after the lost frame.\n");

            printf("Retransmitting frames:\n");

            for (i = lost; i < end; i++)
                printf("Frame %d retransmitted\n", i);
        }
        else if (lost != -1)
        {
            printf("Invalid frame number. Assuming no loss.\n");
        }

        printf("Acknowledgements received for frames %d to %d\n",
               base, end - 1);

        base = end;
    }

    printf("\nAll frames transmitted successfully.\n");

    return 0;
}
