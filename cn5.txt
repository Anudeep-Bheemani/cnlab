#include <stdio.h>

int main()
{
    int n, i, j, start;
    int graph[10][10];
    int visited[10] = {0};
    int queue[10];
    int front = 0, rear = 0;

    printf("Enter number of hosts: ");
    scanf("%d", &n);

    if (n < 1 || n > 10)
    {
        printf("Invalid number of hosts.\n");
        return 1;
    }

    printf("Enter adjacency matrix:\n");

    for (i = 0; i < n; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &graph[i][j]);
        }
    }

    printf("Enter source host (1-%d): ", n);
    scanf("%d", &start);

    if (start < 1 || start > n)
    {
        printf("Invalid source host.\n");
        return 1;
    }

    start = start - 1;

    visited[start] = 1;
    queue[rear++] = start;

    printf("\nBroadcast Tree:\n");

    while (front < rear)
    {
        int current = queue[front++];

        for (i = 0; i < n; i++)
        {
            if (graph[current][i] == 1 && visited[i] == 0)
            {
                visited[i] = 1;
                queue[rear++] = i;

                printf("H%d -> H%d\n", current + 1, i + 1);
            }
        }
    }

    printf("\nHosts not reachable from source H%d:\n", start + 1);

    for (i = 0; i < n; i++)
    {
        if (visited[i] == 0)
        {
            printf("H%d\n", i + 1);
        }
    }

    return 0;
}
